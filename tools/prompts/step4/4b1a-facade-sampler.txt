export async function loadFacade(file): doc = await io.read(file). Walk doc.getRoot().listScenes()[0] nodes recursively (node.listChildren()), world matrix = node.getWorldMatrix() (column-major 16 array). For each node.getMesh() primitive: POSITION (prim.getAttribute('POSITION').getElement(i, arr) returns floats, normalized accessors already decoded), TEXCOORD_0 likewise, indices prim.getIndices().getArray() (or 0..n-1 if none). Transform positions by the world matrix. Material: prim.getMaterial().getBaseColorTexture() -> texture.getImage() (encoded PNG/JPEG/WebP bytes) -> sharp(Buffer.from(img)).ensureAlpha().raw().toBuffer({ resolveWithObject: true }) -> { data, info.width, info.height } (cache per texture). UV transform: const ti = mat.getBaseColorTextureInfo(); const tt = ti && ti.getExtension('KHR_texture_transform'); if tt: offset = tt.getOffset(), scale = tt.getScale() (rotation ignored), uv' = uv * scale + offset. Return { tris: Float32Array (9 floats per triangle, world space), uvs: Float32Array (6 per triangle, transformed), tex: Int32Array (texture index per triangle, -1 if none), textures: [{ data, width, height }], bbox: { min: [x,y,z], max: [x,y,z] } }.
